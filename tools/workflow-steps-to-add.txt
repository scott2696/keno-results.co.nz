# Paste these three steps into .github/workflows/main.yml, between the
# "Rebuild pages" step and the "Commit if anything changed" step.
#
# Also add a repository secret named RALFY_API_KEY
#   Settings -> Secrets and variables -> Actions -> New repository secret
# CI has no macOS keychain, so without it the indexing step skips.
#
# Indentation matters: six spaces before "- name:", eight before its keys.
# These are the same commands tools/refresh.sh already runs locally, and each
# is continue-on-error, so an editorial or indexing failure can never stop the
# draw results being published.

      - name: Watch for lottery news and official releases
        continue-on-error: true
        run: |
          python3 tools/watch_news.py     || true
          python3 tools/fetch_official.py || true

      - name: Write any article the data justifies
        continue-on-error: true
        run: |
          python3 tools/auto_news.py || true
          python3 tools/make_art.py  || true
          python3 tools/build.py > /dev/null

      - name: Submit new URLs for indexing
        env:
          RALFY_API_KEY: ${{ secrets.RALFY_API_KEY }}
        continue-on-error: true
        run: python3 tools/submit_index.py
